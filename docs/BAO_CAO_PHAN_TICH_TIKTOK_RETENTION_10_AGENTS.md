### 1.3. Mô Hình Toán Học Đo Lường Drop-Off Rate & Inter-Part Retention

Trong chuỗi sản xuất 3 tập ($P_1 \to P_2 \to P_3$), các đại lượng suy hao được xác định chính xác theo công thức:
- **Tỷ lệ suy hao bước (Drop-off Rate)**: 
  $$\text{Drop-off}_{A \to B} = \frac{\text{Views}(A) - \text{Views}(B)}{\text{Views}(A)} \times 100\%$$
- **Tỷ lệ giữ chân chuyển tập (Inter-Part Retention Rate)**:
  $$\text{Retention}_{A \to B} = \frac{\text{Views}(B)}{\text{Views}(A)} \times 100\% = 100\% - \text{Drop-off}_{A \to B}$$

### 5.3. Giải Pháp Kỹ Thuật Tự Động Hóa Playwright & FFmpeg Engine

#### A. Playwright Automation: Tự Động Gán & Tạo Mới Tuyển Tập Trên TikTok Studio (`scripts/tiktok_publisher_core.py`)
```python
from playwright.sync_api import Page

def assign_video_to_playlist(page: Page, playlist_name: str) -> bool:
    """
    Tự động hóa chọn hoặc tạo mới Tuyển tập trên form Upload của TikTok Studio.
    Bảo vệ tự động: Hỗ trợ Selector song ngữ Anh - Việt, chờ động qua wait_for()
    và cơ chế Graceful Fallback khi tài khoản (ví dụ 2.5K followers) chưa mở khóa tính năng.
    """
    try:
        # 1. Selector song ngữ EN/VI tìm dropdown Danh sách phát / Playlist
        dropdown = page.locator(
            'div[role="combobox"]:has-text("danh sách phát"), '
            'div[role="combobox"]:has-text("Tuyển tập"), '
            'div[role="combobox"]:has-text("Playlist"), '
            'div[role="combobox"]:has-text("Series")'
        ).first

        # Kiểm tra tính khả dụng của tính năng Playlist trên tài khoản
        if dropdown.count() == 0 or not dropdown.is_visible():
            print("  ⚠️ [Playlist Safeguard] Kênh (2.5K follower) chưa mở khóa tính năng Tuyển tập TikTok Studio.")
            print("  ➡️ Tự động kích hoạt Phễu Fallback: Chuyển hướng sang Bình luận ghim + Unique Series Hashtag!")
            return False

        dropdown.click()
        # Chờ menu options hiển thị động thay vì sleep tĩnh
        page.locator('div[role="listbox"], div[role="menu"]').first.wait_for(state="visible", timeout=4000)

        # 2. Tìm Tuyển tập đã tồn tại theo tên bộ truyện
        target_item = page.locator(f'div[role="option"]:has-text("{playlist_name}")').first
        if target_item.count() > 0 and target_item.is_visible():
            target_item.click()
            print(f"  ✅ Đã gán thành công vào Tuyển tập có sẵn: '{playlist_name}'")
            return True

        # 3. Nếu chưa có, bấm nút Tạo mới (Hỗ trợ song ngữ EN/VI)
        create_btn = page.locator(
            'button:has-text("Tạo danh sách phát"), '
            'div:has-text("Tạo danh sách phát mới"), '
            'button:has-text("Create playlist"), '
            'div:has-text("Create new playlist")'
        ).first

        if create_btn.count() > 0 and create_btn.is_visible():
            create_btn.click()
            dialog_input = page.locator('div[role="dialog"] input').first
            dialog_input.wait_for(state="visible", timeout=4000)
            dialog_input.fill(playlist_name[:30])

            save_btn = page.locator(
                'div[role="dialog"] button:has-text("Tạo"), '
                'div[role="dialog"] button:has-text("Lưu"), '
                'div[role="dialog"] button:has-text("Create"), '
                'div[role="dialog"] button:has-text("Save")'
            ).first
            save_btn.click()
            print(f"  🎉 Đã tạo mới và gán thành công Tuyển tập: '{playlist_name}'")
            return True

        return False
    except Exception as e:
        print(f"  ⚠️ [Playlist Warning] Không thể cấu hình Playlist ({e}). Kích hoạt Phễu Fallback.")
        return False
```

#### B. FFmpeg Filter: Tự Động Nhúng End-Screen Card Trong 8 Giây Cuối (`core/ffmpeg_utils.py`)
```python
from core.ffmpeg_utils import get_vietnamese_font_path

def build_end_screen_filter(duration_sec: float, part_num: int) -> str:
    """
    Tự động nhúng đồ họa End-Screen Card trong 8 giây cuối video.
    Sử dụng hàm get_vietnamese_font_path() động hỗ trợ đa nền tảng (Windows/Linux/Docker),
    khắc phục hoàn toàn lỗi văng crash khi hardcode đường dẫn font Windows.
    """
    start_t = max(0.0, duration_sec - 8.0)
    end_t = duration_sec
    line1 = f"DON XEM TAP {part_num + 1} TIEP THEO TAI TUYEN TAP" if part_num < 3 else "TRON BO 3 TAP DA HOAN TAT - BAM FOLLOW KENH"
    line2 = "Hoac bam vao Binh luan ghim de xem tiep!" if part_num < 3 else "De lai binh luan cam nghi cua ban ben duoi nhe!"
    
    # Lấy đường dẫn font hệ thống tự động tương thích 100% tiếng Việt
    font = get_vietnamese_font_path() or "arial"
    box = (
        f"drawbox=x=160:y=1400:w=760:h=180:color=black@0.8:t=fill:enable='between(t,{start_t},{end_t})',"
        f"drawbox=x=160:y=1400:w=760:h=180:color=0xFFD700@0.9:t=4:enable='between(t,{start_t},{end_t})'"
    )
    t1 = f"drawtext=fontfile='{font}':text='>> {line1} <<':fontsize=32:fontcolor=0xFFD700:x=(w-text_w)/2:y=1430:enable='between(t,{start_t},{end_t})'"
    t2 = f"drawtext=fontfile='{font}':text='{line2}':fontsize=26:fontcolor=white:x=(w-text_w)/2:y=1500:enable='between(t,{start_t},{end_t})'"
    return f"{box},{t1},{t2}"
```
